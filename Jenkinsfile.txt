pipeline {

    agent any

    stages {

        stage('Clone Repository') {
            steps {
                echo 'Repository already connected through Jenkins SCM'
            }
        }

        stage('Build') {
            steps {
                bat 'mvn clean package'
            }
        }
    }

    post {

        success {
            echo 'Build Successful'
        }

        failure {
            echo 'Build Failed'
        }
    }
}
